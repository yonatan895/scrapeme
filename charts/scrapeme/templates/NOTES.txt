ACCESSING THE APPLICATION
-------------------------

1. Get the metrics endpoint:
{{- if .Values.ingress.enabled }}
   URL: http{{ if .Values.ingress.tls }}s{{ end }}://{{ (index .Values.ingress.hosts 0).host }}/metrics
{{- else if contains "NodePort" .Values.service.type }}
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "scrapeme.fullname" . }})
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo "Metrics URL: http://$NODE_IP:$NODE_PORT/metrics"
{{- else if contains "LoadBalancer" .Values.service.type }}
   NOTE: It may take a few minutes for the LoadBalancer IP to be available.
   kubectl get svc --namespace {{ .Release.Namespace }} -w {{ include "scrapeme.fullname" . }}
{{- else if contains "ClusterIP" .Values.service.type }}
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "scrapeme.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
   echo "Metrics URL: http://127.0.0.1:{{ .Values.service.port }}/metrics"
{{- end }}

2. Check health status:
   kubectl get pods -n {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "scrapeme.name" . }}"

3. View logs:
   kubectl logs -n {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "scrapeme.name" . }}" -f
